local function kp(t, c)
	return ColorSequenceKeypoint.new(t, c)
end
local function scaleC(c, f)
	return Color3.new(math.clamp(c.R * f, 0, 1), math.clamp(c.G * f, 0, 1), math.clamp(c.B * f, 0, 1))
end
local function buildSeq(kind, c)
	if kind == "btn" then
		return ColorSequence.new({ kp(0, scaleC(c, 1.35)), kp(0.2, scaleC(c, 1.15)), kp(0.6, c), kp(1, scaleC(c, 0.8)) })
	elseif kind == "title" then
		return ColorSequence.new({ kp(0, scaleC(c, 1.1)), kp(0.3, scaleC(c, 0.95)), kp(1, scaleC(c, 0.8)) })
	elseif kind == "small" then
		return ColorSequence.new({ kp(0, scaleC(c, 1.2)), kp(0.35, c), kp(1, scaleC(c, 0.85)) })
	end
	return ColorSequence.new(scaleC(c, 0.45))
end

local function create(class, props, parent)
	local inst = Instance.new(class)
	for k, v in pairs(props) do
		inst[k] = v
	end
	if parent then
		inst.Parent = parent
	end
	return inst
end

local function corner(parent, r)
	return create("UICorner", { CornerRadius = UDim.new(0, r or 3) }, parent)
end

local function stroke(parent, color, th, transp)
	return create("UIStroke", {
		Color = color,
		Thickness = th or 1,
		Transparency = transp or 0,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, parent)
end

local function safeCallback(callback, ...)
	if type(callback) ~= "function" then
		return
	end
	local args = table.pack(...)
	task.spawn(function()
		local ok, err = pcall(callback, table.unpack(args, 1, args.n))
		if not ok then
			warn("[Cindora] Callback error: " .. tostring(err))
		end
	end)
end

local function deepCopy(v)
	if type(v) == "table" then
		local t = {}
		for k, x in pairs(v) do
			t[k] = deepCopy(x)
		end
		return t
	end
	return v
end

local function copyToClipboard(str)
	local fn = setclipboard or toclipboard or (syn and syn.write_clipboard) or (Clipboard and Clipboard.set)
	if fn then
		pcall(fn, str)
		return true
	end
	return false
end

local function colorToHex(c)

local function hexToColor(str)
	local hex = (tostring(str):gsub("#", ""))
	if #hex == 6 then
		local r, g, b = tonumber(hex:sub(1, 2), 16), tonumber(hex:sub(3, 4), 16), tonumber(hex:sub(5, 6), 16)
		if r and g and b then
			return Color3.fromRGB(r, g, b)
		end
	end
	return nil
end

local function validateKey(K, input)
	if K.Api then
		local ok, body = pcall(function()
			local url = string.gsub(K.Api, "{key}", function()
				return HttpService:UrlEncode(input)
			end)
			return game:HttpGet(url)
		end)
		if not ok or type(body) ~= "string" then
			return false
		end
		local okJ, data = pcall(function()
			return HttpService:JSONDecode(body)
		end)
		if okJ and type(data) == "table" then
			return data.valid == true or data.success == true
		end
		return string.lower(body):match("^%s*true%s*$") ~= nil
	end
	local k = K.Key
	if type(k) == "function" then
		local ok, res = pcall(k, input)
		return ok and res == true
	elseif type(k) == "table" then
		return table.find(k, input) ~= nil
	end
	return input == k
end

local function keyFile(K)
	return K.Folder .. "/key_" .. tostring(lp.UserId) .. ".json"
end

local function saveVerification(K, remember, entered)
	pcall(function()
		if not (isfolder and makefolder and writefile) then
			return
		end
		local file = keyFile(K)
		if not remember then
			if isfile and delfile and isfile(file) then
				delfile(file)
			end
			return
		end
		if not isfolder(K.Folder) then
			makefolder(K.Folder)
		end
		writefile(file, HttpService:JSONEncode({
			userId = lp.UserId,
			timestamp = os.time(),
			key = entered,
		}))
	end)
end

local function readSavedKey(K)
	local ok, key, ts = pcall(function()
		local file = keyFile(K)
		if not (isfile and readfile) or not isfile(file) then
			return nil
		end
		local data = HttpService:JSONDecode(readfile(file))
		if type(data) ~= "table" or data.userId ~= lp.UserId or type(data.key) ~= "string" then
			return nil
		end
		local elapsed = os.time() - (tonumber(data.timestamp) or 0)
		if elapsed >= 0 and elapsed < K.ExpirySeconds then
			return data.key, tonumber(data.timestamp)
		end
		return nil
	end)
	if ok then
		return key, ts
	end
	return nil
end

local function showSplash(opts, parent, S)
	local title = S.Title or opts.Name
	local subtitle = S.Subtitle or ""
	local hold = S.Hold or 4

	local splash = create("ScreenGui", {
		Name = "Cindora_Splash",
		IgnoreGuiInset = true,
		DisplayOrder = 999999,
		ResetOnSpawn = false,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
	})
	local bg = create("Frame", {
		Size = UDim2.new(1, 0, 1, 0),
		BackgroundColor3 = BLACK,
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ZIndex = 999999,
	}, splash)
	local center = create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.new(0.5, 0, 0.48, 0),
		Size = UDim2.new(0.9, 0, 0.9, 0),
		BackgroundTransparency = 1,
		ZIndex = 1000000,
	}, bg)
	create("UIListLayout", {
		HorizontalAlignment = Enum.HorizontalAlignment.Center,
		VerticalAlignment = Enum.VerticalAlignment.Center,
		Padding = UDim.new(0, 16),
		SortOrder = Enum.SortOrder.LayoutOrder,
	}, center)

	local logo = makeLogo(center, opts.Logo, UDim2.new(0.38, 0, 0.46, 0), nil, 1000001, 1)
	Instance.new("UIAspectRatioConstraint", logo).AspectRatio = 1

	create("TextLabel", {
		Text = title,
		FontFace = FBoldItalic,
		TextSize = 50,
		TextColor3 = WHITE,
		TextTransparency = 1,
		BackgroundTransparency = 1,
		Size = UDim2.new(1, 0, 0, 56),
		ZIndex = 1000001,
		LayoutOrder = 2,
	}, center)
	create("TextLabel", {
		Text = subtitle,
		FontFace = FMedium,
		TextSize = 26,
		TextColor3 = Color3.fromRGB(150, 150, 158),
		TextTransparency = 1,
		BackgroundTransparency = 1,
		Size = UDim2.new(1, 0, 0, 32),
		ZIndex = 1000001,
		LayoutOrder = 3,
	}, center)

	local targets = {}
	for _, inst in ipairs(center:GetDescendants()) do
		if inst:IsA("ImageLabel") then
			table.insert(targets, { inst, "ImageTransparency" })
		elseif inst:IsA("TextLabel") then
			table.insert(targets, { inst, "TextTransparency" })
		end
	end
	for _, t in ipairs(targets) do
		t[1][t[2]] = 1
	end

	splash.Parent = parent

	local function fade(visible, info)
		local first = TweenService:Create(bg, info, { BackgroundTransparency = visible and 0 or 1 })
		first:Play()
		for _, t in ipairs(targets) do
			TweenService:Create(t[1], info, { [t[2]] = visible and 0 or 1 }):Play()
		end
		return first
	end

	fade(true, TweenInfo.new(1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)).Completed:Wait()
	task.wait(hold)
	fade(false, TweenInfo.new(1, Enum.EasingStyle.Quad, Enum.EasingDirection.In)).Completed:Wait()
	splash:Destroy()
end

		end
	end

-- Bloqueia até a key ser validada. Retorna ok, timestamp.
local function runKeyGate(opts, parent)
	local K = opts.Key
	if K.SkipIfSaved then
		local saved, ts = readSavedKey(K)
		if saved and validateKey(K, saved) then
			return true, ts or os.time()
		end
	end
	local done, okResult, tsResult = false, false, nil
	showKeyMenu(opts, parent, function(ok, ts)
		okResult, tsResult, done = ok, ts, true
	end)
	while not done do
		task.wait(0.1)
	end
	return okResult, tsResult
end

		local c = typeof(accent) == "Color3" and accent or Cindora.AccentColors[accent]
		if not c then
			return
		end
		for kind in pairs(CurSeqs) do
			CurSeqs[kind] = buildSeq(kind, c)
		end
		for i = #Themed, 1, -1 do
			local t = Themed[i]
			if t.grad.Parent then
				t.grad.Color = CurSeqs[t.kind]
			else
				table.remove(Themed, i)
			end
		end

	local function makeDraggable(handle, target, clamp)
		local dragging, dragStart, startPos
		handle.InputBegan:Connect(function(input)
			if isPress(input) then
				dragging = true
				dragStart = input.Position
				startPos = target.Position
				input.Changed:Connect(function()
					if input.UserInputState == Enum.UserInputState.End then
						dragging = false
					end
				end)
			end
		end)
		connect(UIS.InputChanged, function(input)
			if dragging and isMove(input) then
				local d = input.Position - dragStart
				local px, py = startPos.X.Offset + d.X, startPos.Y.Offset + d.Y
				if clamp then
					local vs = workspace.CurrentCamera.ViewportSize
					local cx = math.clamp(startPos.X.Scale * vs.X + px, 60, math.max(61, vs.X - 60))
					local cy = math.clamp(startPos.Y.Scale * vs.Y + py, 30, math.max(31, vs.Y - 30))
					px = cx - startPos.X.Scale * vs.X
					py = cy - startPos.Y.Scale * vs.Y
				end
				target.Position = UDim2.new(startPos.X.Scale, px, startPos.Y.Scale, py)
			end
		end)
	end

	function Window:SetTheme(t)
		local th = type(t) == "table" and t or Cindora.Themes[t]
		if not th then
			return
		end
		local old = Theme
		Theme = { Main = th.Main or BLACK, Row = th.Row or BLACK, Text = th.Text or TEXT }
		main.BackgroundColor3 = Theme.Main
		searchPanel.BackgroundColor3 = Theme.Main
		for _, d in ipairs(gui:GetDescendants()) do
			if d ~= main and d ~= searchPanel and d:IsA("GuiObject") then
				if d.BackgroundColor3 == old.Row then
					d.BackgroundColor3 = Theme.Row
				end
				if (d:IsA("TextLabel") or d:IsA("TextButton") or d:IsA("TextBox")) and d.TextColor3 == old.Text then
					d.TextColor3 = Theme.Text
				end
			end
		end
		if th.Accent then
			Window:SetAccent(th.Accent)
			local ao = Window.Options._uiAccent
			if ao and type(th.Accent) == "string" then
				ao:Set(th.Accent, true)
			end
		end
	end

	local function applyFilter()
		local q = string.lower(searchBox.Text)
		for _, f in ipairs(Features) do
			local match = true
			if f.page == (currentTab and currentTab.Page) then
				match = (q == "") or (string.find(string.lower(f.name), q, 1, true) ~= nil)
			end
			f.holder.Visible = match and not f.hidden
		end
	end
	searchBox:GetPropertyChangedSignal("Text"):Connect(applyFilter)

	local function switchTab(tab)
		for _, t in ipairs(Window.Tabs) do
			t.Page.Visible = (t == tab)
			local s = TabStrokes[t]
			if s then
				tw(s, 0.15, { Color = (t == tab) and Color3.fromRGB(235, 235, 240) or Color3.fromRGB(70, 70, 78) })
			end
		end
		currentTab = tab
		tab.Page.Position = UDim2.fromOffset(0, TITLE_H + u(14))
		tw(tab.Page, 0.3, { Position = UDim2.fromOffset(0, TITLE_H) }, Enum.EasingStyle.Quint)
		applyFilter()
	end

	local tipToken, tipActive = 0, false
	local function addTip(obj, text)
		if not UIS.MouseEnabled then
			return
		end
		obj.MouseEnter:Connect(function()
			tipToken += 1
			local mine = tipToken
			task.delay(0.4, function()
				if mine == tipToken then
					tipLabel.Text = text
					tipFrame.Visible = true
					tipActive = true
				end
			end)
		end)
		obj.MouseLeave:Connect(function()
			tipToken += 1
			tipFrame.Visible = false
			tipActive = false
		end)
	end
	connect(RunService.RenderStepped, function()
		if not tipActive then
			return
		end
		local m = UIS:GetMouseLocation()
		local vs = workspace.CurrentCamera.ViewportSize
		local sz = tipFrame.AbsoluteSize
		tipFrame.Position = UDim2.fromOffset(
			math.clamp(m.X + 14, 4, math.max(5, vs.X - sz.X - 4)),
			math.clamp(m.Y + 18, 4, math.max(5, vs.Y - sz.Y - 4))
		)
	end)

	local function registerFlag(flag, getter, setter, default, ctrl)
		if flag then
			Registry[flag] = { get = getter, set = setter, default = deepCopy(default) }
			if ctrl then
				Window.Options[flag] = ctrl
			end
		end
	end

	local Keybinds = {}
	local capture

	local function finish(ctrl, holder, cname, page, flag, getter, setter, default)
		local entry = { holder = holder, name = cname, page = page, hidden = false }
		recolor(holder)
		table.insert(Features, entry)
		ctrl.Holder = holder
		function ctrl:SetVisible(v)
			entry.hidden = not v
			applyFilter()
		end
		function ctrl:Destroy()
			holder:Destroy()
			local i = table.find(Features, entry)
			if i then table.remove(Features, i) end
			if flag then
				Registry[flag] = nil
				Window.Options[flag] = nil
			end
		end
		if flag then
			registerFlag(flag, getter, setter, default, ctrl)
		end
		return ctrl
	end

	local PILL_STROKE_ON = Color3.fromRGB(120, 120, 128)
	local KNOB_ON = Color3.fromRGB(18, 18, 22)
	local PILL_STROKE_OFF = Color3.fromRGB(10, 10, 10)

	local function makePill(parent, state, onChange)
		state = state and true or false
		local pill = create("TextButton", {
			AnchorPoint = Vector2.new(1, 0.5),
			Position = UDim2.new(1, -u(3.5), 0.5, 0),
			Size = UDim2.fromOffset(u(34), u(15.5)),
			BackgroundColor3 = WHITE,
			AutoButtonColor = false,
			Text = "",
		}, parent)
		corner(pill, 3)
		local pst = stroke(pill, state and PILL_STROKE_ON or PILL_STROKE_OFF, 1)
		local grad = gradient(pill, state and GREEN_ON or GRAY_OFF, 90)

		local knobW, knobH = u(11.5), u(11)
		local function knobPos(on)
			return on and UDim2.new(1, -(knobW + u(3)), 0.5, 0) or UDim2.new(0, u(3), 0.5, 0)
		end
		local knob = create("Frame", {
			AnchorPoint = Vector2.new(0, 0.5),
			Position = knobPos(state),
			Size = UDim2.fromOffset(knobW, knobH),
			BackgroundColor3 = state and KNOB_ON or WHITE,
			BorderSizePixel = 0,
		}, pill)
		corner(knob, 2)

		local P = { Frame = pill }
		function P:Set(v, silent)
			state = v and true or false
			grad.Color = state and GREEN_ON or GRAY_OFF
			pst.Color = state and PILL_STROKE_ON or PILL_STROKE_OFF
			tw(knob, 0.18, { Position = knobPos(state), BackgroundColor3 = state and KNOB_ON or WHITE }, Enum.EasingStyle.Back)
			if not silent and onChange then
				onChange(state)
			end
		end
		function P:Get()
			return state
		end
		pill.MouseButton1Click:Connect(function()
			P:Set(not state)
		end)
		hoverScale(pill, 1.05, 0.93)
		return P
	end

		-- Toggle
		function Sec:AddToggle(o)
			local height = o.Desc and u(27) or u(21)
			local ts = o.TitleSize or (o.Desc and Sec.TitleSize - 1.2 or Sec.TitleSize)
			local kb = nil
			if type(o.Keybind) == "table" then
				kb = o.Keybind
			elseif o.Keybind == true then
				kb = {}
			end
			local keyReserve = kb and (u(30) + u(5)) or 0
			local holder, row = makeRow(container, nextOrder(), height, o.Sub and u(13) or 0, o.Sub and u(12) or 0)
			rowText(row, o.Name, o.Desc, height, o.Sub and u(5) or u(7), u(50) + keyReserve, ts)
			local P = makePill(row, o.Default or false, function(v)
				safeCallback(o.Callback, v)
			end)
			local T = {}
			function T:Set(v, silent) P:Set(v, silent) end
			function T:Get() return P:Get() end

			if kb then
				local key = kb.Default
				if type(key) == "string" then
					local okk, kc = pcall(function() return Enum.KeyCode[key] end)
					key = okk and kc or nil
				end
				local kbox = create("TextButton", {
					AnchorPoint = Vector2.new(1, 0.5),
					Position = UDim2.new(1, -(u(3.5) + u(34) + u(4)), 0.5, 0),
					Size = UDim2.fromOffset(u(30), u(13)),
					BackgroundColor3 = BLACK,
					BackgroundTransparency = 0.3,
					AutoButtonColor = false,
					FontFace = FBold,
					Text = key and key.Name or "None",
					TextColor3 = WHITE,
					TextSize = u(5.5),
					TextTruncate = Enum.TextTruncate.AtEnd,
				}, row)
				corner(kbox, 2)
				stroke(kbox, BLACK, 1, 0.4)
				hoverScale(kbox, 1.05, 0.95)

				function T:SetKey(k, silent)
					if type(k) == "string" then
						local okk, kc = pcall(function() return Enum.KeyCode[k] end)
						k = okk and kc or nil
					end
					key = (typeof(k) == "EnumItem" and k.EnumType == Enum.KeyCode) and k or nil
					kbox.Text = key and key.Name or "None"
					kbox.TextColor3 = WHITE
					if not silent then safeCallback(kb.Callback, key) end
				end
				function T:GetKey() return key end

				kbox.MouseButton1Click:Connect(function()
					kbox.Text = "..."
					kbox.TextColor3 = GREEN_T
					capture = function(k)
						if k == Enum.KeyCode.Escape then
							T:SetKey(key, true)
						elseif k == Enum.KeyCode.Backspace then
							T:SetKey(nil)
						else
							T:SetKey(k)
						end
					end
				end)

				table.insert(Keybinds, {
					get = function() return key end,
					pressed = function() P:Set(not P:Get()) end,
					holder = holder,
				})
				registerFlag(kb.Flag, function() return key and key.Name or "" end, function(v) T:SetKey(v) end, key and key.Name or "")
			end

			return finish(T, holder, o.Name, page, o.Flag, function() return P:Get() end, function(v) P:Set(v) end, P:Get())
		end

		-- Slider
		function Sec:AddSlider(o)
			local min, max = o.Min or 0, o.Max or 100
			if max < min then min, max = max, min end
			if max == min then max = min + 1 end
			local value = math.clamp(o.Default or min, min, max)
			local suffix = o.Suffix or ""
			local decimals = o.Decimals or 0
			local mult = 10 ^ decimals
			local full = o.Sub ~= true
			local inset = full and 0 or u(13)
			local insetR = full and 0 or u(12)

			local holder, row = makeRow(container, nextOrder(), u(21), inset, insetR)
			local trackW = full and u(56) or u(50)
			local trackMarginR = u(6)
			local valueW = full and u(46) or u(30)
			local labelX = u(full and 7 or 5)

			create("TextLabel", {
				BackgroundTransparency = 1,
				Position = UDim2.fromOffset(labelX, 0),
				Size = UDim2.new(1, -(labelX + trackW + trackMarginR + valueW + u(6)), 0, u(21)),
				FontFace = FBold,
				Text = o.Name,
				TextColor3 = TEXT,
				TextSize = u(full and 10 or 8.5),
				TextXAlignment = Enum.TextXAlignment.Left,
				TextTruncate = Enum.TextTruncate.AtEnd,
			}, row)
			local valueLabel = create("TextBox", {
				ClearTextOnFocus = false,
				BackgroundTransparency = 1,
				AnchorPoint = Vector2.new(1, 0),
				Position = UDim2.new(1, -(trackW + trackMarginR + u(3)), 0, 0),
				Size = UDim2.fromOffset(valueW, u(21)),
				FontFace = FBold,
				Text = "",
				TextColor3 = GREEN_T,
				TextSize = u(7),
				TextXAlignment = Enum.TextXAlignment.Right,
			}, row)
			local track = create("TextButton", {
				AnchorPoint = Vector2.new(1, 0.5),
				Position = UDim2.new(1, -trackMarginR, 0.5, 0),
				Size = UDim2.fromOffset(trackW, u(8)),
				BackgroundColor3 = Color3.fromRGB(26, 26, 19),
				AutoButtonColor = false,
				Text = "",
			}, row)
			corner(track, 2)
			local fill = create("Frame", {
				Size = UDim2.new(0, 0, 1, 0),
				BackgroundColor3 = WHITE,
				BorderSizePixel = 0,
			}, track)
			corner(fill, 2)
			gradient(fill, GREEN_FILL, 90)
			local knobW = u(8)
			local knob = create("Frame", {
				AnchorPoint = Vector2.new(0, 0.5),
				Size = UDim2.fromOffset(knobW, u(10.5)),
				BackgroundColor3 = WHITE,
				BorderSizePixel = 0,
				ZIndex = 2,
			}, track)
			corner(knob, 1)

			local dragging = false
			local function render(animate)
				local a = (value - min) / (max - min)
				local x = math.floor(a * (trackW - knobW))
				local fillSize = UDim2.new(0, x + knobW * 0.5, 1, 0)
				local knobPos = UDim2.new(0, x, 0.5, 0)
				if animate and not dragging then
					tw(fill, 0.15, { Size = fillSize })
					tw(knob, 0.15, { Position = knobPos })
				else
					fill.Size = fillSize
					knob.Position = knobPos
				end
				valueLabel.Text = string.format("%." .. decimals .. "f", value) .. suffix
			end

			local Sl = {}
			function Sl:Set(v, silent)
				v = tonumber(v)
				if not v then return end
				v = math.clamp(v, min, max)
				v = math.floor(v * mult + 0.5) / mult
				value = v
				render(true)
				if not silent then
					safeCallback(o.Callback, value)
				end
			end
			function Sl:Get() return value end

			local function update(input)
				local a = math.clamp((input.Position.X - track.AbsolutePosition.X - knobW / 2) / math.max(1, track.AbsoluteSize.X - knobW), 0, 1)
				Sl:Set(min + (max - min) * a)
			end
			track.InputBegan:Connect(function(input)
				if isPress(input) then
					dragging = true
					update(input)
				end
			end)
			connect(UIS.InputEnded, function(input)
				if isPress(input) then dragging = false end
			end)
			connect(UIS.InputChanged, function(input)
				if dragging and isMove(input) then update(input) end
			end)

			render(false)
			valueLabel.Focused:Connect(function()
				valueLabel.Text = tostring(value)
			end)
			valueLabel.FocusLost:Connect(function()
				local n = tonumber(valueLabel.Text)
				if n then
					Sl:Set(n)
				else
					render(false)
				end
			end)

			return finish(Sl, holder, o.Name, page, o.Flag, function() return value end, function(v) Sl:Set(v) end, value)
		end

			}, box)

				Position = UDim2.fromOffset(0, rowH + u(2)),
				Size = UDim2.new(1, 0, 0, 0),
				BackgroundColor3 = BLACK,
				BackgroundTransparency = 0.3,
				BorderSizePixel = 0,
				ClipsDescendants = true,
			}, holder)

			}, panel)

			stroke(search, BLACK, 1, 0.3)

				Position = UDim2.fromOffset(u(4), gridTop),
				Size = UDim2.new(1, -u(8), 0, 0),
				BackgroundTransparency = 1,
				BorderSizePixel = 0,
				ScrollBarThickness = math.max(2, u(2)),

				CellSize = UDim2.new(1 / COLS, -math.ceil(gap * (COLS - 1) / COLS), 0, cellH),
				SortOrder = Enum.SortOrder.LayoutOrder,
			}, grid)

			local selected = {}
			local single = o.Default
			local cells = {}
			local panelH = 0
			local isOpenDD = false
			local D = {}

			local function currentText()
				if multi then
					local n = 0
					for _ in pairs(selected) do n += 1 end
					if n == 0 then return o.EmptyText or "None" end
					return n .. " selected"
				end
				return single ~= nil and tostring(single) or "None"
			end

			local function refreshItems()
				for opt, c in pairs(cells) do
					local on = (multi and selected[opt]) or (not multi and single == opt)
					tw(c.ind, 0.12, { BackgroundColor3 = on and GREEN_T or Color3.fromRGB(22, 22, 22) })
				end
				valueLabel.Text = currentText()
			end

			local function relayout()
				local visible = 0
				local q = string.lower(search.Text)
				for _, opt in ipairs(options) do
					local c = cells[opt]
					if c then
						local show = q == "" or string.find(string.lower(tostring(opt)), q, 1, true) ~= nil
						c.btn.Visible = show
						if show then visible += 1 end
					end
				end
				local rows = math.ceil(visible / COLS)
				local gridH = rows > 0 and math.min(rows * cellH + (rows - 1) * gap, maxGridH) or 0
				grid.Size = UDim2.new(1, -u(8), 0, gridH)
				panelH = gridTop + gridH + u(3)
				if isOpenDD then
					tw(holder, 0.15, { Size = UDim2.new(1, 0, 0, rowH + u(2) + panelH) })
					tw(panel, 0.15, { Size = UDim2.new(1, 0, 0, panelH) })
				end
			end

			local function toggleList(v)
				isOpenDD = v
				if v then
					search.Text = ""
					relayout()
				else
					tw(holder, 0.2, { Size = UDim2.new(1, 0, 0, rowH) })
					tw(panel, 0.2, { Size = UDim2.new(1, 0, 0, 0) })
				end
				tw(arrowLbl, 0.2, { Rotation = v and 180 or 0 })
			end

			function D:Get()
				if multi then
					local t = {}
					for _, opt in ipairs(options) do
						if selected[opt] then t[#t + 1] = opt end
					end
					return t
				end
				return single
			end

			function D:Set(v, silent)
				if multi then
					selected = {}
					for _, opt in ipairs(v or {}) do
						if table.find(options, opt) then selected[opt] = true end
					end
				else
					single = v
				end
				refreshItems()
				if not silent then
					safeCallback(o.Callback, D:Get())
				end
			end

			local function build()
				for _, c in pairs(cells) do
					c.btn:Destroy()
				end
				cells = {}
				for i, opt in ipairs(options) do
					local b = create("TextButton", {
						LayoutOrder = i,
						BackgroundColor3 = BLACK,
						BackgroundTransparency = 0.35,
						AutoButtonColor = false,
						Text = "",
					}, grid)
					corner(b, 2)
					local ind = create("Frame", {
						AnchorPoint = Vector2.new(0, 0.5),
						Position = UDim2.new(0, u(3), 0.5, 0),
						Size = UDim2.fromOffset(u(4), u(7)),
						BackgroundColor3 = Color3.fromRGB(22, 22, 22),
						BorderSizePixel = 0,
					}, b)
					corner(ind, 1)
					create("TextLabel", {
						BackgroundTransparency = 1,
						Position = UDim2.fromOffset(u(11), 0),
						Size = UDim2.new(1, -u(13), 1, 0),
						FontFace = FBold,
						Text = tostring(opt),
						TextColor3 = WHITE,
						TextSize = u(5.8),
						TextXAlignment = Enum.TextXAlignment.Left,
						TextTruncate = Enum.TextTruncate.AtEnd,
					}, b)
					cells[opt] = { btn = b, ind = ind }
					b.MouseEnter:Connect(function() tw(b, 0.1, { BackgroundTransparency = 0.15 }) end)
					b.MouseLeave:Connect(function() tw(b, 0.1, { BackgroundTransparency = 0.35 }) end)
					b.MouseButton1Click:Connect(function()
						if multi then
							selected[opt] = (not selected[opt]) or nil
							refreshItems()
							safeCallback(o.Callback, D:Get())
						else
							single = opt
							refreshItems()
							toggleList(false)
							safeCallback(o.Callback, single)
						end
					end)
				end
				relayout()
			end

			function D:Refresh(newOptions, newValue)
				options = type(newOptions) == "table" and newOptions or {}
				build()
				if newValue ~= nil then
					D:Set(newValue, true)
				elseif not multi and single ~= nil and not table.find(options, single) then
					D:Set(nil, true)
				elseif multi then
					D:Set(D:Get(), true)
				else
					refreshItems()
				end
			end

			search:GetPropertyChangedSignal("Text"):Connect(relayout)
			build()
			D:Set(deepCopy(o.Default), true)
			box.MouseButton1Click:Connect(function() toggleList(not isOpenDD) end)
			hoverScale(box, 1.04, 0.96)

			return finish(D, holder, o.Name, page, o.Flag, function() return D:Get() end, function(v) D:Set(v) end, deepCopy(o.Default))
		end

				local c = Color3.fromHSV(h, s, v)
				swatch.BackgroundColor3 = c
				sv.BackgroundColor3 = Color3.fromHSV(h, 1, 1)
				svCursor.Position = UDim2.new(s, 0, 1 - v, 0)
				hueCursor.Position = UDim2.new(0, 0, h, 0)
				if not silent then
					safeCallback(o.Callback, c)
				end
			end

			local inset = game:GetService("GuiService"):GetGuiInset().Y
			local dragSV, dragH = false, false
			local function updSV(input)
				s = math.clamp((input.Position.X - sv.AbsolutePosition.X) / math.max(1, sv.AbsoluteSize.X), 0, 1)
				v = 1 - math.clamp((input.Position.Y + inset - sv.AbsolutePosition.Y) / math.max(1, sv.AbsoluteSize.Y), 0, 1)
				apply()
			end
			local function updH(input)
				h = math.clamp((input.Position.Y + inset - hue.AbsolutePosition.Y) / math.max(1, hue.AbsoluteSize.Y), 0, 1)
				apply()
			end
			svInput.InputBegan:Connect(function(input)
				if isPress(input) then
					dragSV = true
					updSV(input)
				end
			end)
			hueInput.InputBegan:Connect(function(input)
				if isPress(input) then
					dragH = true
					updH(input)
				end
			end)
			connect(UIS.InputEnded, function(input)
				if isPress(input) then
					dragSV, dragH = false, false
				end
			end)
			connect(UIS.InputChanged, function(input)
				if isMove(input) then
					if dragSV then
						updSV(input)
					elseif dragH then
						updH(input)
					end
				end
			end)

			local openCP = false
			swatch.MouseButton1Click:Connect(function()
				openCP = not openCP
				tw(holder, 0.2, { Size = UDim2.new(1, 0, 0, openCP and (rowH + u(2) + PANEL_H) or rowH) })
				tw(panel, 0.2, { Size = UDim2.new(1, 0, 0, openCP and PANEL_H or 0) })
			end)

			local CP = {}
			function CP:Get() return Color3.fromHSV(h, s, v) end
			function CP:Set(c, silent)
				if type(c) == "string" then c = hexToColor(c) end
				if typeof(c) ~= "Color3" then return end
				h, s, v = c:ToHSV()
				apply(silent)
			end

			apply(true)
			return finish(CP, holder, o.Name, page, o.Flag, function() return colorToHex(CP:Get()) end, function(val) CP:Set(val) end, colorToHex(start))
		end

		-- Proteção contra erros + dicas (Tip = "...") em todos os componentes
		local function guard(fname)
			local orig = Sec[fname]
			local textOk = (fname == "AddInfo" or fname == "AddLabel" or fname == "AddParagraph" or fname == "AddDivider")
			Sec[fname] = function(self, o, ...)
				if not textOk and type(o) ~= "table" then
					warn("[Cindora] " .. fname .. " expects a table, e.g. { Name = \"...\" }")
					return dummyCtrl()
				end
				local okG, res = pcall(orig, self, o, ...)
				if not okG then
					warn("[Cindora] " .. fname .. " failed: " .. tostring(res))
					return dummyCtrl()
				end
				if type(o) == "table" and o.Tip and type(res) == "table" and res.Holder then
					addTip(res.Holder, tostring(o.Tip))
				end
				return res
			end
		end
		for _, fname in ipairs({ "AddInfo", "AddLabel", "AddToggle", "AddSlider", "AddDropdown", "AddInput", "AddButton", "AddKeybind", "AddColorPicker", "AddParagraph", "AddDivider" }) do
			guard(fname)
		end

		return Sec
	end

	-- Configs (salvar / carregar)
	------------------------------------------------------------
	local Configs = {}
	local DIR = folder .. "/Configs"
	local MAX_CFGS = opts.MaxConfigs or 10
	local hasFS = isfolder and makefolder and writefile and readfile and listfiles and delfile
	if hasFS then
		pcall(function()
			if not isfolder(folder) then makefolder(folder) end
			if not isfolder(DIR) then makefolder(DIR) end
		end)
	end

	function Window:GetConfigs()
		local seen, t = {}, {}
		for n in pairs(Configs) do seen[n] = true end
		if hasFS then
			pcall(function()
				for _, f in ipairs(listfiles(DIR)) do
					local n = tostring(f):gsub("\\", "/"):match("([^/]+)%.json$")
					if n then seen[n] = true end
				end
			end)
		end
		for n in pairs(seen) do t[#t + 1] = n end
		table.sort(t)
		return t
	end

	local function readCfg(cname)
		if Configs[cname] then return Configs[cname] end
		if hasFS then
			local ok, data = pcall(function()
				return HttpService:JSONDecode(readfile(DIR .. "/" .. cname .. ".json"))
			end)
			if ok and type(data) == "table" then return data end
		end
		return nil
	end

	function Window:SaveConfig(cname)
		cname = string.sub((string.gsub(tostring(cname or ""), "[^%w_%-]", "")), 1, 32)
		if cname == "" then cname = "default" end
		local names = Window:GetConfigs()
		if not table.find(names, cname) and #names >= MAX_CFGS then
			return false, string.format(L.ConfigLimit, MAX_CFGS), cname
		end
		local snap = {}
		for flag, f in pairs(Registry) do
			snap[flag] = deepCopy(f.get())
		end
		Configs[cname] = snap
		if hasFS then
			pcall(function()
				writefile(DIR .. "/" .. cname .. ".json", HttpService:JSONEncode(snap))
			end)
		end
		return true, string.format(L.ConfigSaved, cname, #Window:GetConfigs(), MAX_CFGS), cname
	end

	function Window:LoadConfig(cname)
		local snap = cname and readCfg(cname)
		if not snap then
			return false, L.ConfigNone
		end
		for flag, v in pairs(snap) do
			if Registry[flag] then Registry[flag].set(deepCopy(v)) end
		end
		return true, string.format(L.ConfigLoaded, cname)
	end

	function Window:DeleteConfig(cname)
		if not cname then
			return false, L.ConfigNone
		end
		Configs[cname] = nil
		if hasFS then
			pcall(function() delfile(DIR .. "/" .. cname .. ".json") end)
		end
		return true, string.format(L.ConfigDeleted, cname)
	end

	function Window:ResetToDefaults()
		for _, f in pairs(Registry) do
			f.set(deepCopy(f.default))
		end
		return true, L.ConfigReset
	end

	local AUTOFILE = folder .. "/autoload.txt"

	function Window:SetAutoLoad(cname)
		if not hasFS then return false end
		pcall(function()
			if cname and cname ~= "" then
				writefile(AUTOFILE, cname)
			elseif isfile and isfile(AUTOFILE) then
				delfile(AUTOFILE)
			end
		end)
		return true
	end

	function Window:GetAutoLoad()
		local okA, v = pcall(function()
			if hasFS and isfile and isfile(AUTOFILE) then
				return readfile(AUTOFILE)
			end
		end)
		if okA and type(v) == "string" and v ~= "" then
			return v
		end
		return nil
	end

	function Window:LoadAutoConfig(silent)
		local n = Window:GetAutoLoad()
		if not n then return false end
		local okL = Window:LoadConfig(n)
		if okL and not silent then
			Window:Notify("Config", "Auto loaded: " .. n, 2.5, "success")
		end
		return okL
	end

		})
		ui:AddSlider({
			Name = "UI Transparency", Min = 0, Max = 0.8, Default = Settings.Transparency, Decimals = 2, Flag = "_uiTransparency",
			Callback = function(v)
				Settings.Transparency = v
				main.BackgroundTransparency = v
				searchPanel.BackgroundTransparency = math.max(0, v - 0.2)
			end,
		})
		local themeNames = { "Dark" }
		local extraThemes = {}
		for n in pairs(Cindora.Themes) do
			if n ~= "Dark" then extraThemes[#extraThemes + 1] = n end
		end
		table.sort(extraThemes)
		for _, n in ipairs(extraThemes) do themeNames[#themeNames + 1] = n end
		ui:AddDropdown({
			Name = "Theme", Options = themeNames, Default = opts.Theme or "Dark", Flag = "_uiTheme", Search = false,
			Callback = function(v) Window:SetTheme(v) end,
		})

		local accentNames = { "Black", "Red", "Orange", "Green", "Blue", "Purple", "Pink" }
		local extra = {}
		for n in pairs(Cindora.AccentColors) do
			if not table.find(accentNames, n) then extra[#extra + 1] = n end
		end
		table.sort(extra)
		for _, n in ipairs(extra) do accentNames[#accentNames + 1] = n end
		ui:AddDropdown({
			Name = "Accent Color", Options = accentNames, Default = opts.Accent or "Black", Flag = "_uiAccent",
			Callback = function(v) Window:SetAccent(v) end,
		})

		local mgr = cfgTab:AddSection("Configs")
		local cfgName = "default"
		mgr:AddInput({
			Name = "Config Name", Default = cfgName, Placeholder = "name",
			Callback = function(v) cfgName = v end,
		})
		local cfgDropdown = mgr:AddDropdown({ Name = "Saved Configs", Options = Window:GetConfigs() })
		mgr:AddButton({ Name = "Save Config", Callback = function()
			local ok, msg, saved = Window:SaveConfig(cfgName)
			if ok then cfgDropdown:Refresh(Window:GetConfigs(), saved) end
			Window:Notify("Config", msg, 2.5)
		end })
		mgr:AddButton({ Name = "Load Selected", Callback = function()
			local _, msg = Window:LoadConfig(cfgDropdown:Get())
			Window:Notify("Config", msg, 2.5)
		end })
		mgr:AddButton({ Name = "Delete Selected", Callback = function()
			local ok, msg = Window:DeleteConfig(cfgDropdown:Get())
			if ok then cfgDropdown:Refresh(Window:GetConfigs(), nil) end
			Window:Notify("Config", msg, 2.5)
		end })
		mgr:AddButton({ Name = "Reset To Defaults", Callback = function()
			local _, msg = Window:ResetToDefaults()
			Window:Notify("Config", msg, 2.5)
		end })

		mgr:AddButton({ Name = "Set As Auto Load", Callback = function()
			local sel = cfgDropdown:Get()
			if not sel then
				Window:Notify("Config", L.ConfigNone, 2.5, "warning")
				return
			end
			Window:SetAutoLoad(sel)
			Window:Notify("Config", "Auto load: " .. sel, 2.5, "success")
		end })
		mgr:AddButton({ Name = "Clear Auto Load", Callback = function()
			Window:SetAutoLoad(nil)
			Window:Notify("Config", "Auto load cleared.", 2.5, "info")
		end })

		local share = cfgTab:AddSection("Share Config", true)
		share:AddButton({ Name = "Copy Config To Clipboard", Callback = function()
			if copyToClipboard(Window:ExportConfig()) then
				Window:Notify("Config", L.ExportCopied, 2.5, "success")
			else
				Window:Notify("Config", L.ClipboardMissing, 2.5, "error")
			end
		end })
		local importText = ""
		share:AddInput({ Name = "Paste Config", Placeholder = "paste", Callback = function(v) importText = v end })
		share:AddButton({ Name = "Import Pasted Text", Callback = function()
			local okI, msg = Window:ImportConfig(importText)
			Window:Notify("Config", msg, 2.5, okI and "success" or "error")
		end })
		if getclipboard then
			share:AddButton({ Name = "Import From Clipboard", Callback = function()
				local okC, txt = pcall(getclipboard)
				local okI, msg = Window:ImportConfig(okC and txt or "")
				Window:Notify("Config", msg, 2.5, okI and "success" or "error")
			end })
		end

		local danger = cfgTab:AddSection("Danger Zone")
		danger:AddButton({ Name = "Destroy UI", Callback = function() Window:Destroy() end })
	end

	------------------------------------------------------------
	-- Eventos finais
	------------------------------------------------------------
	closeBtn.MouseButton1Click:Connect(function()
		Window:SetOpen(false)
		Window:Notify(name, string.format(L.Closed, Settings.ToggleKey.Name), 3)
	end)
	minBtn.MouseButton1Click:Connect(function()
		Window:SetMinimized(not minimized)
	end)

	connect(UIS.InputBegan, function(input, processed)
		if capture and input.UserInputType == Enum.UserInputType.Keyboard then
			local f = capture
			capture = nil
			f(input.KeyCode)
			return
		end
		if processed or input.UserInputType ~= Enum.UserInputType.Keyboard then return end
		if input.KeyCode == Settings.ToggleKey then
			Window:Toggle()
			return
		end
		for _, kb in ipairs(Keybinds) do
			if kb.pressed and kb.holder.Parent and kb.get() == input.KeyCode then
				safeCallback(kb.pressed)
			end
		end
	end)

	function Window:Destroy()
		for _, c in ipairs(conns) do
			pcall(function() c:Disconnect() end)
		end
		conns = {}
		if gui then gui:Destroy() end
		local i = table.find(Cindora.Windows, Window)
		if i then table.remove(Cindora.Windows, i) end
	end

	if opts.Theme then
		Window:SetTheme(opts.Theme)
	end
	if opts.AutoLoad ~= false then
		task.defer(function()
			task.wait(0.5) -- espera o script do usuário terminar de criar os componentes
			if gui.Parent then
				Window:LoadAutoConfig(true)
			end
		end)
	end
	if opts.Accent then
		Window:SetAccent(opts.Accent)
	end
	table.insert(Cindora.Windows, Window)

	if opts.Notify ~= false then
		Window:Notify(name, string.format(L.Loaded, Settings.ToggleKey.Name), 4)
	end

	return Window
end

return Cindora
